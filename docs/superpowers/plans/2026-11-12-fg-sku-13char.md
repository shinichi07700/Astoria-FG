> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Goal:** Replace the FFS-FPS SKU codification with the 13-char `[Brand3][App2][Form2][Seq3][Size1][Rev2]` scheme, re-code legacy SKUs with a legacy alias, and leave every downstream FK untouched.

**Architecture:** Additive migration 020 (m_brand, dosage_form, revisi, fg_master segments + unique kode_fg). New SKUs use the 13-char code as `id`; legacy rows keep `ffs|fps` ids until recoded. All downstream views render codes through `Store.fgById(...).kodeFG`, so they need no edits. Dual-path store logic keeps legacy data working until recode.

**Tech Stack:** Zero-dependency vanilla JS (ES5 style: `var`, `function`, IIFE namespaces on `window`), Supabase PostgreSQL (manual SQL migrations), node-run unit tests (`node tests/*.test.js`), GitHub Pages deploy.

**Spec:** `docs/superpowers/specs/2026-11-12-fg-sku-13char-design.md`

```js
  /* ---------- 13-char SKU codification (2026-11-12 scheme) ---------- */
  var APP_MAP = {
    "Eye Makeup": "EY", "Face Makeup": "FC", "Lip Makeup": "LP", "Skincare": "SK",
    "Cleansers & Makeup Removers": "CL", "Sun Care": "SN", "Body Care": "BD", "Hair Care": "HC"
  };
  var DOSAGE_FORMS = [
    ["LQ", "Liquid"], ["CR", "Cream"], ["SS", "Semi-Solid"], ["LP", "Loose Powder"],
    ["CP", "Compact Powder"], ["LS", "Lipstick"], ["ST", "Stick"]
  ];
  var SKU_RE = /^[A-Z]{3}(EY|FC|LP|SK|CL|SN|BD|HC)(LQ|CR|SS|LP|CP|LS|ST)\d{3}[A-Z]\d{2}$/;
  function appOf(kategori) { return APP_MAP[kategori] || ""; }
  function skuCode(s) {
    return (s.brand || "") + (s.app || "") + (s.form || "") +
      pad(s.seq || 0, 3) + (s.size || "") + pad(s.rev || 0, 2);
  }
  function validSkuCode(code) { return SKU_RE.test(String(code || "")); }
  /* Seq: reuse the (brand,app,form,formula) group's seq, else max+1 in (brand,app,form). 0 = overflow. */
  function seqFor(rows, brand, app, form, formulaId) {
    var group = rows.filter(function (f) {
      return f.brandId === brand && f.app === app && f.form === form && f.formulaId === formulaId && f.seq;
    });
    if (group.length) return group[0].seq;
    var max = 0;
    rows.forEach(function (f) {
      if (f.brandId === brand && f.app === app && f.form === form && f.seq > max) max = f.seq;
    });
    return max >= 999 ? 0 : max + 1;
  }
  /* Size: same netto reuses its letter; new netto gets highest+1. "" = overflow. */
  function sizeFor(rows, brand, app, form, seq, netto) {
    var own = "", max = 0;
    rows.forEach(function (f) {
      if (f.brandId === brand && f.app === app && f.form === form && f.seq === seq && f.sizeLetter) {
        var c = f.sizeLetter.charCodeAt(0) - 64;
        if (c > max) max = c;
        if (Number(f.nettoMl) === Number(netto)) own = f.sizeLetter;
      }
    });
    if (own) return own;
    return max >= 26 ? "" : String.fromCharCode(65 + max);
  }
```

  /* Formula revision: same issued code, swap the formula link, clear NA. */
  function reviseFormulaFG(oldId, newFormulaId) {
    var old = fgById(oldId), f = formulaById(newFormulaId);
    if (!old || !f) return null;
    if (!f.dosageForm) return { error: "form" };
    if (Engine.appOf(f.kategori) !== old.app || f.dosageForm !== old.form) return { error: "category" };
    old.formulaId = newFormulaId;
    old.ffs = newFormulaId;
    old.kodeNA = ""; old.tglExpire = "";
    saveFG(old, false, oldId);
    audit("REVISION", "Master F/G", old.kodeFG + " formula revised to " + newFormulaId + " - NA cleared for BPOM recertification");
    save();
    return old;
  }
  /* Packaging revision: new SKU at higher FPS revisi, same netto; old superseded. */
  function revisePackagingFG(oldId, newPackagingId) {
    var old = fgById(oldId), pack = packagingById(newPackagingId);
    if (!old || !pack) return null;
    if (Number(pack.volumeMl) !== Number(old.nettoMl)) return { error: "netto" };
    if ((Number(pack.revisi) || 1) <= old.rev) return { error: "rev" };
    var rec = JSON.parse(JSON.stringify(old));
    rec.packagingId = pack.id; rec.fps = pack.id;
    rec.rev = Number(pack.revisi) || 1;
    rec.kodeNA = ""; rec.tglExpire = ""; rec.discontinue = false;
    old.discontinue = true;
    saveFG(old, false, oldId);
    saveFG(rec, true, null);
    audit("REVISION", "Master F/G", old.kodeFG + " superseded by " + rec.kodeFG + " (packaging rev " + rec.rev + ") - NA cleared for BPOM recertification");
    save();
    return rec;
  }
  /* One-time legacy recode: freeze segments, alias the old code, keep the id. */
  function recodeFG(id, seg) {
    var f = fgById(id);
    if (!f) return null;
    f.brandId = seg.brand; f.formulaId = seg.formulaId; f.packagingId = seg.packagingId;
    f.ffs = seg.formulaId; f.fps = seg.packagingId;
    f.app = seg.app; f.form = seg.form; f.seq = seg.seq; f.sizeLetter = seg.size;
    f.nettoMl = seg.netto; f.rev = seg.rev;
    f.legacyCode = f.kodeFG;
    f.kodeFG = Engine.skuCode({ brand: seg.brand, app: seg.app, form: seg.form, seq: seg.seq, size: seg.size, rev: seg.rev });
    f.status = Engine.evaluateStatus(f.formulaId, f.packagingId, f.kodeNA, f.tglExpire, f.discontinue);
    f.diubahOleh = db.meta.user.email || db.meta.user.name;
    f.waktuUpdate = Engine.nowWIB();
    audit("RECODE", "Master F/G", f.legacyCode + " -> " + f.kodeFG + " (" + f.deskripsi + ")");
    if (window.Sync) Sync.mark("fg_master", f.id);
    save();
    return f;
  }
